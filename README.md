**Use your preferred IDE**

If you want to work locally using your own IDE, you can clone this repo and push changes. Pushed changes will also be reflected in Lovable.

The only requirement is having Node.js & npm installed - [install with nvm](https://github.com/nvm-sh/nvm#installing-and-updating)

Follow these steps:

```sh

git clone <YOUR_GIT_URL>

# Step 2: Navigate to the project directory.
cd <YOUR_PROJECT_NAME>

# Step 3: Install the necessary dependencies.
npm i

# Step 4: Start the development server with auto-reloading and an instant preview.
npm run dev
```

**Edit a file directly in GitHub**

- Navigate to the desired file(s).
- Click the "Edit" button (pencil icon) at the top right of the file view.
- Make your changes and commit the changes.

**Use GitHub Codespaces**

- Navigate to the main page of your repository.
- Click on the "Code" button (green button) near the top right.
- Select the "Codespaces" tab.
- Click on "New codespace" to launch a new Codespace environment.
- Edit files directly within the Codespace and commit and push your changes once you're done.

Create the table in Supabase SQL editor:

```sql
create table if not exists public.present_students (
  id bigint generated by default as identity primary key,
  student_id text not null,
  cohort text not null check (cohort in ('B','C')),
  timestamp timestamptz not null default now()
);

alter table public.present_students enable row level security;
create policy "Allow read" on public.present_students for select using (true);
create policy "Allow insert" on public.present_students for insert with check (true);

-- Create cancelled_sessions table for tracking class cancellations
create table if not exists public.cancelled_sessions (
  id bigint generated by default as identity primary key,
  date date not null,
  cohort text not null check (cohort in ('B','C')),
  is_cancelled boolean not null default true,
  created_at timestamptz not null default now(),
  unique(date, cohort)
);

alter table public.cancelled_sessions enable row level security;
create policy "Allow read" on public.cancelled_sessions for select using (true);
create policy "Allow insert" on public.cancelled_sessions for insert with check (true);
create policy "Allow update" on public.cancelled_sessions for update using (true);

-- Create class_schedule table to define which days of the week classes occur
create table if not exists public.class_schedule (
  id bigint generated by default as identity primary key,
  cohort text not null check (cohort in ('B','C')),
  day_of_week integer not null check (day_of_week >= 0 and day_of_week <= 6), -- 0 = Sunday, 1 = Monday, etc.
  unique(cohort, day_of_week)
);

alter table public.class_schedule enable row level security;
create policy "Allow read" on public.class_schedule for select using (true);
create policy "Allow insert" on public.class_schedule for insert with check (true);
create policy "Allow update" on public.class_schedule for update using (true);
create policy "Allow delete" on public.class_schedule for delete using (true);

-- Create class_dates table to track actual dates when classes occurred
-- This allows overriding the schedule for holidays, make-up classes, etc.
create table if not exists public.class_dates (
  id bigint generated by default as identity primary key,
  date date not null,
  cohort text not null check (cohort in ('B','C')),
  unique(date, cohort)
);

alter table public.class_dates enable row level security;
create policy "Allow read" on public.class_dates for select using (true);
create policy "Allow insert" on public.class_dates for insert with check (true);
create policy "Allow update" on public.class_dates for update using (true);
create policy "Allow delete" on public.class_dates for delete using (true);

-- Note: After creating these tables, you should:
-- 1. Use the admin dashboard "Class Schedule" button to configure which days classes occur (typically 3 days per week)
-- 2. Use "Generate Class Dates" to automatically create class dates based on the schedule
-- 3. Attendance will only be tracked on days when classes actually occurred

-- Create students table if it doesn't exist (or ensure it has the required structure)
create table if not exists public.students (
  student_id text primary key,
  cohort text not null check (cohort in ('B','C')),
  name text
);

alter table public.students add column if not exists name text;
```
